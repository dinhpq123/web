Ví dụ tích hợp
curl — Lấy trạng thái trạm bơm
curl -u demo:iclev \
  https://tools.iclever.vn/hadiwa-simulator/api/admin/status
curl — Đổi kịch bản
curl -u demo:iclev -X POST \
  -H "Content-Type: application/json" \
  -d '{"scenario":"flood"}' \
  https://tools.iclever.vn/hadiwa-simulator/api/admin/scenario
JavaScript — SSE real-time
const es = new EventSource(
  '/hadiwa-simulator/api/admin/stream',
  { withCredentials: true }
);
es.onmessage = e => {
  const data = JSON.parse(e.data);
  console.log(data);
};
MQTT — Subscribe dữ liệu
mqtt.connect('tcp://tools.iclever.vn:1883',
  { clientId: 'my-system' }
).subscribe('hadiwa/pump/#')
 .subscribe('hadiwa/flood/#');
